```python
# core_backtest.py (nuevo módulo)
def backtest(ticker: str, historico: pd.DataFrame, fundamentales_por_fecha: dict,
             fecha_inicio: date, fecha_fin: date, paso_dias: int = 30) -> list[dict]:
    resultados = []
    for fecha in rango_fechas(fecha_inicio, fecha_fin, paso_dias):
        hist_recortado = historico[historico.index <= fecha]        # sin look-ahead
        fund = fundamentales_por_fecha.get(fecha_mas_reciente_disponible(fecha))
        ind = core_indicadores.calcular(hist_recortado)
        conf = core_confluencia.detectar(hist_recortado, ind)
        fv = core_fair_value.calcular(fund, precio=hist_recortado["close"].iloc[-1])
        calidad = core_calidad.metricas(fund)
        timing = core_timing.evaluar(ind, conf, calidad)
        plan = core_plan_dca.plan(conf, ind, hist_recortado["close"].iloc[-1], fv["fair_value"])
        ver = core_plan_dca.veredicto(calidad["nota"], fv["upside_pct"], timing["nota"])
        resultados.append({"fecha": fecha, "veredicto": ver, "plan": plan,
                            "precio_real_a_90d": precio_futuro(historico, fecha, dias=90)})
    return resultados  # comparar ver["clave"] == "comprar" contra precio_real_a_90d
```

Esto permitiría validar cuantitativamente, con datos históricos reales, si el propio método del comité (triple condición + veredicto con vetos) habría funcionado — más allá de la revisión caso a caso que hacemos hoy.
