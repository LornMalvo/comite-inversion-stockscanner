    limite = LIMITE_TELEGRAM - margen

    actual = ""

        candidato = f"{actual}\n\n{bloque}" if actual else bloque
        if len(candidato) > limite:
            if actual:
                partes.append(actual)
            actual = bloque
        else:
            actual = candidato
    if actual:

    for parte in partes:
